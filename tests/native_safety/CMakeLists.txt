cmake_minimum_required(VERSION 3.15)
project(MaestroNativeSafety LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(MAESTRO_WINDOWS_PROBE_ONLY "Build only the standalone E02a Windows SDK probe" OFF)
if(WIN32)
    add_executable(maestro_windows_security windows_security.cpp
        ../../lbug-src/src/common/file_system/root_directory_windows_security.cpp
        ../../lbug-src/src/common/exception/exception.cpp)
    target_include_directories(maestro_windows_security PRIVATE
        ../../lbug-src/src/include ../../lbug-src/src/common/file_system)
    target_compile_definitions(maestro_windows_security PRIVATE
        MAESTRO_WINDOWS_SECURITY_TEST LBUG_STATIC_DEFINE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(maestro_windows_security PRIVATE advapi32)
    enable_testing()
    # Only the focused workflow executes security tests: all live handles must be
    # opened by its verified standard-user child, not the full-engine admin runner.
    if(MAESTRO_WINDOWS_PROBE_ONLY)
        add_test(NAME maestro_windows_security COMMAND maestro_windows_security
            "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/windows-private-root-v1.tsv")
        set_tests_properties(maestro_windows_security PROPERTIES TIMEOUT 120)
    endif()
endif()
if(MAESTRO_WINDOWS_PROBE_ONLY)
    if(NOT WIN32)
        message(FATAL_ERROR "MAESTRO_WINDOWS_PROBE_ONLY requires the Windows SDK")
    endif()
    add_executable(maestro_windows_capability_probe windows_capability_probe.cpp)
    execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
        OUTPUT_VARIABLE MAESTRO_PROBE_SHA OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE MAESTRO_SHA_RESULT)
    if(NOT MAESTRO_SHA_RESULT EQUAL 0)
        message(FATAL_ERROR "Cannot bind the probe to its source revision")
    endif()
    target_compile_definitions(maestro_windows_capability_probe PRIVATE
        _WIN32_WINNT=0x0A00 WINVER=0x0A00
        MAESTRO_WINDOWS_SDK="${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}"
        MAESTRO_PROBE_SHA="${MAESTRO_PROBE_SHA}")
    enable_testing()
    add_test(NAME maestro_windows_capability_probe COMMAND maestro_windows_capability_probe
        "${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(maestro_windows_capability_probe PROPERTIES TIMEOUT 120)
    return()
endif()
set(BUILD_SHELL OFF CACHE BOOL "" FORCE)
set(BUILD_SINGLE_FILE_HEADER OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LBUG OFF CACHE BOOL "" FORCE)
set(AUTO_UPDATE_GRAMMAR OFF CACHE BOOL "" FORCE)
set(LBUG_EXTENSION_INSTALLER OFF CACHE BOOL "" FORCE)
add_subdirectory(../../lbug-src engine)
# Only this native harness enables the deterministic check/open interleave seam.
target_compile_definitions(lbug_file_system PRIVATE MAESTRO_NATIVE_OPEN_TEST)
if(WIN32)
    add_executable(maestro_windows_root windows_root.cpp)
    target_link_libraries(maestro_windows_root PRIVATE lbug advapi32)
    target_compile_definitions(maestro_windows_root PRIVATE MAESTRO_NATIVE_OPEN_TEST _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    enable_testing()
    add_test(NAME maestro_windows_root COMMAND maestro_windows_root "${CMAKE_CURRENT_BINARY_DIR}")
    set_tests_properties(maestro_windows_root PROPERTIES TIMEOUT 120)
endif()
add_executable(maestro_native_safety native_safety.cpp)
target_link_libraries(maestro_native_safety PRIVATE lbug)
add_executable(maestro_rooted_sidecars rooted_sidecars.cpp)
target_link_libraries(maestro_rooted_sidecars PRIVATE lbug)
# GNU ld wrapping observes the real directory-sync call, not the pre-call test hook.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(maestro_rooted_sidecars PRIVATE "-Wl,--wrap=fsync")
endif()
add_executable(maestro_rooted_flows rooted_flows.cpp)
target_link_libraries(maestro_rooted_flows PRIVATE lbug libalp)
add_library(maestro_refusal_extension SHARED refusal_extension.cpp)
set_target_properties(maestro_refusal_extension PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_executable(maestro_query_refusals query_refusals.cpp)
target_link_libraries(maestro_query_refusals PRIVATE lbug libalp)
target_include_directories(maestro_query_refusals PRIVATE ../../lbug-src/third_party/thrift)
target_compile_definitions(maestro_query_refusals PRIVATE
    MAESTRO_REFUSAL_EXTENSION="$<TARGET_FILE:maestro_refusal_extension>")
add_dependencies(maestro_query_refusals maestro_refusal_extension)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # Bring std::filesystem into the executable so GNU ld --wrap sees its probes.
    # Test-only, Linux coverage; production keeps its normal C++ runtime linkage.
    target_link_options(maestro_query_refusals PRIVATE "-static-libstdc++")
    target_compile_definitions(maestro_query_refusals PRIVATE MAESTRO_SYSCALL_WRAP)
    foreach(call open openat stat lstat fstat fstatat fopen dlopen mmap)
        target_link_options(maestro_query_refusals PRIVATE "-Wl,--wrap=${call}")
    endforeach()
endif()
enable_testing()
add_test(NAME maestro_query_refusals COMMAND maestro_query_refusals)
add_test(NAME maestro_native_safety COMMAND maestro_native_safety)
add_test(NAME maestro_rooted_sidecars COMMAND maestro_rooted_sidecars)
add_test(NAME maestro_rooted_flows COMMAND maestro_rooted_flows)
set_tests_properties(maestro_rooted_flows maestro_query_refusals PROPERTIES TIMEOUT 120)
